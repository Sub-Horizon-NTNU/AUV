cmake_minimum_required(VERSION 3.20)
project(sonar_simulation VERSION 0.1.0 LANGUAGES CXX)

option(SONAR_BUILD_VIEWER "Build the interactive raylib viewer (downloads pinned raylib)" OFF)

add_library(sonar_core src/simulator.cpp)
target_include_directories(sonar_core PUBLIC include)
target_compile_features(sonar_core PUBLIC cxx_std_17)
set_target_properties(sonar_core PROPERTIES CXX_EXTENSIONS OFF)

add_library(sonar_io src/io.cpp)
target_link_libraries(sonar_io PUBLIC sonar_core)
add_executable(sonar_sim src/main.cpp)
target_link_libraries(sonar_sim PRIVATE sonar_io)

include(CTest)
set(sonar_targets sonar_core sonar_io sonar_sim)
if(SONAR_BUILD_VIEWER)
    enable_language(C)
    include(FetchContent)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(raylib
        GIT_REPOSITORY https://github.com/raysan5/raylib.git
        GIT_TAG 02566453880e06bec97f5b8160fa6fe1e98397ca) # Official 6.2 release.
    FetchContent_MakeAvailable(raylib)
    add_library(sonar_inference visualization/inference.cpp)
    target_link_libraries(sonar_inference PUBLIC sonar_core)
    add_executable(sonar_viewer visualization/main.cpp visualization/model.cpp visualization/view.cpp
        visualization/wavefront.cpp)
    target_link_libraries(sonar_viewer PRIVATE sonar_io sonar_inference raylib)
    list(APPEND sonar_targets sonar_inference sonar_viewer)
endif()
if(BUILD_TESTING)
    add_executable(sonar_tests tests/test_simulator.cpp)
    target_link_libraries(sonar_tests PRIVATE sonar_io)
    list(APPEND sonar_targets sonar_tests)
    add_test(NAME sonar_numerics COMMAND sonar_tests)
    add_test(NAME sonar_cli_help COMMAND sonar_sim --help)
    add_test(NAME sonar_cli_invalid_angle COMMAND sonar_sim --angle-grad 400)
    set_tests_properties(sonar_cli_invalid_angle PROPERTIES WILL_FAIL TRUE)
    if(SONAR_BUILD_VIEWER)
        add_executable(sonar_viewer_tests visualization/test_model.cpp visualization/model.cpp
            visualization/wavefront.cpp)
        target_link_libraries(sonar_viewer_tests PRIVATE sonar_io sonar_inference)
        list(APPEND sonar_targets sonar_viewer_tests)
        add_test(NAME sonar_viewer_model COMMAND sonar_viewer_tests)
        add_executable(sonar_wavefront_tests visualization/test_wavefront.cpp visualization/wavefront.cpp)
        target_link_libraries(sonar_wavefront_tests PRIVATE sonar_core)
        list(APPEND sonar_targets sonar_wavefront_tests)
        add_test(NAME sonar_geometric_wavefront COMMAND sonar_wavefront_tests)
        add_executable(sonar_inference_tests visualization/test_inference.cpp)
        target_link_libraries(sonar_inference_tests PRIVATE sonar_inference)
        list(APPEND sonar_targets sonar_inference_tests)
        add_test(NAME sonar_pool_inference COMMAND sonar_inference_tests)
    endif()
endif()

foreach(target IN LISTS sonar_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endforeach()
