cmake_minimum_required(VERSION 3.20)
project(sonar_simulation VERSION 0.1.0 LANGUAGES CXX)

add_library(sonar_core src/simulator.cpp)
target_include_directories(sonar_core PUBLIC include)
target_compile_features(sonar_core PUBLIC cxx_std_17)
set_target_properties(sonar_core PROPERTIES CXX_EXTENSIONS OFF)

add_library(sonar_io src/io.cpp)
target_link_libraries(sonar_io PUBLIC sonar_core)
add_executable(sonar_sim src/main.cpp)
target_link_libraries(sonar_sim PRIVATE sonar_io)

include(CTest)
set(sonar_targets sonar_core sonar_io sonar_sim)
if(BUILD_TESTING)
    add_executable(sonar_tests tests/test_simulator.cpp)
    target_link_libraries(sonar_tests PRIVATE sonar_io)
    list(APPEND sonar_targets sonar_tests)
    add_test(NAME sonar_numerics COMMAND sonar_tests)
    add_test(NAME sonar_cli_help COMMAND sonar_sim --help)
    add_test(NAME sonar_cli_invalid_angle COMMAND sonar_sim --angle-grad 400)
    set_tests_properties(sonar_cli_invalid_angle PROPERTIES WILL_FAIL TRUE)
endif()

foreach(target IN LISTS sonar_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endforeach()
